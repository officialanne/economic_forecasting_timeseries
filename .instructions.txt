# setup
clone the git repository

# create a virtual environment
python -m venv .venv
.\.venv\Scripts\activate

# install dependencies
pip install -r requirements.txt
pip install ipykernel
python -m pip list

// run the python Scripts
python scripts/[name]

// because the venv is already here
.\.venv\Scripts\activate

# run notebooks or Scripts
notebooks/EDA.ipynb – Explore data and visualize trends.
notebooks/Forecasting.ipynb – Fit models and forecast economic indicators.
